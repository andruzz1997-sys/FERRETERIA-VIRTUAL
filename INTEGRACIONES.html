💳 Integración con Stripe

Para aceptar pagos de tarjeta de crédito de manera segura:


// API Key de Stripe
const STRIPE_PUBLIC_KEY = 'pk_test_tu_clave_publica';

// 1. Cargar librería de Stripe
const script = document.createElement('script');
script.src = 'https://js.stripe.com/v3/';
script.async = true;
document.head.appendChild(script);

// 2. Inicializar Stripe
const stripe = Stripe(STRIPE_PUBLIC_KEY);
const elements = stripe.elements();
const cardElement = elements.create('card');
cardElement.mount('#card-element');

// 3. Procesar pago
document.getElementById('pay-btn').addEventListener('click', async () => {
    const {token} = await stripe.createToken(cardElement);
    
    if (token) {
        fetch('/api/payments', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                stripeToken: token.id,
                amount: getCartTotal() * 100,
                currency: 'usd'
            })
        })
        .then(res => res.json())
        .then(data => console.log('Pago procesado:', data));
    }
});

🏦 Integración con PayPal

Aceptar pagos con PayPal:


// 1. Agregar botón de PayPal
<div id="paypal-button-container"></div>

// 2. Script de PayPal
<script src="https://www.paypal.com/sdk/js?client-id=TU_CLIENT_ID"></script>

// 3. Configurar botón
paypal.Buttons({
    createOrder(data, actions) {
        return actions.order.create({
            purchase_units: [{
                amount: {
                    value: getCartTotal().toString()
                }
            }]
        });
    },
    onApprove(data, actions) {
        return actions.order.capture().then(() => {
            showNotification('✅ Pago con PayPal completado');
            clearCart();
        });
    }
}).render('#paypal-button-container');

📊 Google Analytics

Rastrear comportamiento de usuarios:


// Script de Google Analytics
<script async src="https://www.googletagmanager.com/gtag/js?id=GA_ID"></script>
<script>
  window.dataLayer = window.dataLayer || [];
  function gtag(){dataLayer.push(arguments);}
  gtag('js', new Date());
  gtag('config', 'GA_ID');
</script>

// Eventos personalizados
function trackProductView(productId, productName) {
    gtag('event', 'view_item', {
        items: [{
            item_id: productId,
            item_name: productName,
            price: getProductPrice(productId)
        }]
    });
}

function trackAddToCart(item) {
    gtag('event', 'add_to_cart', {
        items: [{
            item_id: item.id,
            item_name: item.name,
            price: item.price,
            quantity: item.quantity
        }]
    });
}

function trackPurchase(orderId, items, total) {
    gtag('event', 'purchase', {
        transaction_id: orderId,
        value: total,
        items: items
    });
}

📧 Emails Transaccionales (SendGrid)

Enviar confirmaciones por email:


// Backend
const sgMail = require('@sendgrid/mail');
sgMail.setApiKey(SENDGRID_API_KEY);

async function sendOrderConfirmation(email, orderId, items) {
    const htmlContent = `
        <h2>Confirmación de Orden #${orderId}</h2>
        <p>Gracias por tu compra en FerrreWeb</p>
        <h3>Productos:</h3>
        <ul>
            ${items.map(item => `<li>${item.name} x${item.quantity}</li>`).join('')}
        </ul>
        <p><a href="https://ferreweb.com/track/${orderId}">Rastrear Orden</a></p>
    `;

    const msg = {
        to: email,
        from: 'noreply@ferreweb.com',
        subject: `Confirmación de Orden - #${orderId}`,
        html: htmlContent
    };

    return sgMail.send(msg);
}

🔥 Firebase (Realtime Database)

Sincronización en tiempo real:


// Inicializar Firebase
import { initializeApp } from "firebase/app";
import { getDatabase, ref, set, onValue } from "firebase/database";

const firebaseConfig = {
    apiKey: "YOUR_API_KEY",
    authDomain: "ferreweb.firebaseapp.com",
    databaseURL: "https://ferreweb.firebaseio.com",
    projectId: "ferreweb"
};

const app = initializeApp(firebaseConfig);
const database = getDatabase(app);

// Guardar carrito en tiempo real
function syncCartToFirebase(userId, cart) {
    set(ref(database, 'users/' + userId + '/cart'), cart);
}

// Escuchar cambios
onValue(ref(database, 'users/' + userId + '/cart'), (snapshot) => {
    const cartData = snapshot.val();
    cart = cartData || [];
    updateCartUI();
});

// Guardar orden
function saveOrderToFirebase(userId, order) {
    set(ref(database, 'orders/' + order.orderId), order);
}

Usuarios con login:


// Instalación
npm install jsonwebtoken

// Backend - Generar Token
const jwt = require('jsonwebtoken');

function generateToken(userId) {
    return jwt.sign(
        { userId, email: user.email },
        process.env.JWT_SECRET,
        { expiresIn: '7d' }
    );
}

// Frontend - Guardar token
function login(email, password) {
    fetch('/api/login', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({email, password})
    })
    .then(res => res.json())
    .then(data => {
        localStorage.setItem('token', data.token);
        redirectToDashboard();
    });
}

// Frontend - Usar token en requests
function authenticatedFetch(url, options = {}) {
    const token = localStorage.getItem('token');
    return fetch(url, {
        ...options,
        headers: {
            ...options.headers,
            'Authorization': `Bearer ${token}`
        }
    });
}

🔍 Búsqueda Avanzada (Elasticsearch)

Motor de búsqueda potente:


// Backend
const elasticsearch = require('@elastic/elasticsearch');
const client = new elasticsearch.Client({ node: 'http://localhost:9200' });

// Indexar productos
async function indexProducts(products) {
    for (const product of products) {
        await client.index({
            index: 'products',
            id: product.id.toString(),
            body: product
        });
    }
}

// Buscar
async function searchProducts(query) {
    const result = await client.search({
        index: 'products',
        body: {
            query: {
                multi_match: {
                    query: query,
                    fields: ['name^2', 'description', 'category']
                }
            }
        }
    });
    return result.hits.hits.map(hit => hit._source);
}
